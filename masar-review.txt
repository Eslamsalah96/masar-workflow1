import "server-only";
import { SignJWT, jwtVerify } from "jose";
import { cookies } from "next/headers";
import { redirect } from "next/navigation";
import { prisma } from "./prisma";

const COOKIE_NAME = "masar_session";

const secretValue = process.env.AUTH_SECRET;

if (!secretValue || secretValue.length < 32) {
  throw new Error("AUTH_SECRET must be at least 32 characters");
}

const secret = new TextEncoder().encode(secretValue);

type SessionUser = {
  id: string;
  role: string;
  departmentId: string | null;
};

export async function createSession(user: SessionUser) {
  const token = await new SignJWT({
    role: user.role,
    departmentId: user.departmentId,
  })
    .setProtectedHeader({ alg: "HS256" })
    .setSubject(user.id)
    .setIssuedAt()
    .setExpirationTime("8h")
    .sign(secret);

  const cookieStore = await cookies();

  cookieStore.set(COOKIE_NAME, token, {
    httpOnly: true,
    secure: process.env.NODE_ENV === "production",
    sameSite: "lax",
    path: "/",
    maxAge: 60 * 60 * 8,
  });
}

export async function deleteSession() {
  const cookieStore = await cookies();
  cookieStore.delete(COOKIE_NAME);
}

export async function getCurrentUser() {
  const cookieStore = await cookies();
  const token = cookieStore.get(COOKIE_NAME)?.value;

  if (!token) return null;

  try {
    const { payload } = await jwtVerify(token, secret);

    if (!payload.sub) return null;

    const user = await prisma.user.findUnique({
      where: { id: payload.sub },
      select: {
        id: true,
        name: true,
        email: true,
        role: true,
        departmentId: true,
        isActive: true,
        department: {
          select: { name: true },
        },
      },
    });

    if (!user || !user.isActive) return null;

    return user;
  } catch (error) {
    console.error("Session verification failed:", error);
    return null;
  }
}

export async function requireUser() {
  const user = await getCurrentUser();

  if (!user) {
    redirect("/login");
  }

  return user;
}

import "server-only";
import { redirect } from "next/navigation";
import { getCurrentUser } from "./auth";

export const ROLES = {
  GENERAL_MANAGER: "GENERAL_MANAGER",
  DEPARTMENT_MANAGER: "DEPARTMENT_MANAGER",
  EMPLOYEE: "EMPLOYEE",
  SECRETARY: "SECRETARY",
  CUSTOMER_SERVICE: "CUSTOMER_SERVICE",
} as const;

// التحقق من تسجيل الدخول والدور
export async function requireRole(allowedRoles: string[]) {
  const user = await getCurrentUser();

  if (!user) {
    redirect("/login");
  }

  if (!allowedRoles.includes(user.role)) {
    redirect("/unauthorized");
  }

  return user;
}

// المدير العام فقط
export async function requireGeneralManager() {
  return requireRole([ROLES.GENERAL_MANAGER]);
}

// المدير العام ومديرو الأقسام
// تستخدم في الصفحات الإدارية العامة، وليس مراجعة الطلبات
export async function requireDepartmentManager() {
  return requireRole([
    ROLES.GENERAL_MANAGER,
    ROLES.DEPARTMENT_MANAGER,
  ]);
}

// الموظفون والمديرون والمدير العام
export async function requireEmployee() {
  return requireRole([
    ROLES.GENERAL_MANAGER,
    ROLES.DEPARTMENT_MANAGER,
    ROLES.EMPLOYEE,
  ]);
}

// صلاحية تسجيل الطلبات
export async function requireRequestRegistration() {
  return requireRole([
    ROLES.GENERAL_MANAGER,
    ROLES.SECRETARY,
    ROLES.CUSTOMER_SERVICE,
  ]);
}

// مدير إدارة الإجراءات فقط
export async function requireProceduresManager() {
  const user = await requireRole([
    ROLES.DEPARTMENT_MANAGER,
  ]);

  if (user.department?.name !== "الإجراءات") {
    redirect("/unauthorized");
  }

  return user;
}

// التحقق من الوصول إلى قسم معين
export async function canAccessDepartment(departmentId: string) {
  const user = await getCurrentUser();

  if (!user) return false;

  if (user.role === ROLES.GENERAL_MANAGER) {
    return true;
  }

  return user.departmentId === departmentId;
}

function getApplicability(payload: Payload): string {
  if (payload.customerType === "EXISTING") return "EXISTING";

  return payload.companyAlreadyEstablished
    ? "NEW_ESTABLISHED"
    : "NEW_UNESTABLISHED";
}

function parseAllowedOptions(optionsJson: string | null): string[] {
  if (!optionsJson) return [];

  try {
    const parsed = JSON.parse(optionsJson);

    if (!Array.isArray(parsed)) return [];

    return parsed.flatMap((item) => {
      if (typeof item === "string") return [item];

      if (item && typeof item === "object") {
        const values = [
          typeof item.label === "string" ? item.label : "",
          typeof item.value === "string" ? item.value : "",
        ];

        return values.filter(Boolean);
      }

      return [];
    });
  } catch {
    return [];
  }
}

function requirementMatchesAnswers(
  requirement: {
    conditions: {
      expectedValue: string;
      groupKey: string;
      question: { code: string };
    }[];
  },
  answers: Record<string, string>
): boolean {
  const conditions = requirement.conditions || [];

  if (conditions.length === 0) return true;

  const groups = new Map<
    string,
    typeof conditions
  >();

  for (const condition of conditions) {
    const key = condition.groupKey || "DEFAULT";
    const current = groups.get(key) || [];

    current.push(condition);
    groups.set(key, current);
  }

  return Array.from(groups.values()).some((group) =>
    group.every(
      (condition) =>
        answers[condition.question.code] === condition.expectedValue
    )
  );
}
export async function GET(request: NextRequest) {
  try {
    const user = await getCurrentUser();

    if (!user || !user.isActive) {
      return NextResponse.json(
        { error: "يجب تسجيل الدخول أولاً" },
        { status: 401 }
      );
    }

    let accessFilter = {};

    if (user.role === "GENERAL_MANAGER") {
      accessFilter = {};
    } else if (user.role === "DEPARTMENT_MANAGER") {
      if (!user.departmentId) {
        return NextResponse.json(
          { error: "المستخدم غير مرتبط بقسم" },
          { status: 403 }
        );
      }

      accessFilter = {
        currentDepartmentId: user.departmentId,
      };
    } else if (user.role === "EMPLOYEE") {
  accessFilter = {
    tasks: {
      some: {
        assignedToId: user.id,
      },
    },
  };
} else if (
  user.role === "SECRETARY" ||
  user.role === "CUSTOMER_SERVICE"
) {
  accessFilter = {
    createdById: user.id,
  };
} else {
      return NextResponse.json(
        { error: "ليس لديك صلاحية عرض قائمة الطلبات" },
        { status: 403 }
      );
    }

    const searchParams = request.nextUrl.searchParams;
    const search = searchParams.get("q")?.trim() || "";
    const status = searchParams.get("status") || "";

    const requests = await prisma.request.findMany({
      where: {
        ...accessFilter,
        ...(status ? { status } : {}),
        ...(search
          ? {
              OR: [
                {
                  requestNumber: {
                    contains: search,
                  },
                },
                {
                  serviceName: {
                    contains: search,
                  },
                },
                {
                  customer: {
                    name: {
                      contains: search,
                    },
                  },
                },
              ],
            }
          : {}),
      },
      include: {
        customer: true,
        createdBy: {
  select: {
    id: true,
    name: true,
    email: true,
    role: true,
  },
},
        currentDepartment: true,
        _count: {
          select: {
            tasks: true,
          },
        },
      },
      orderBy: {
        createdAt: "desc",
      },
      take: 100,
    });

    return NextResponse.json({ requests });

      { status: 500 }

  }

export async function POST(request: NextRequest) {
  let createdRequestId: string | null = null;
  let createdCustomerId: string | null = null;

  const savedFiles: string[] = [];

  try {
    const user = await getCurrentUser();

    if (!user || !user.isActive) {
      return NextResponse.json(
        { error: "يجب تسجيل الدخول أولاً" },
        { status: 401 }
      );
    }

    if (!ALLOWED_ROLES.includes(user.role)) {
      return NextResponse.json(
        { error: "ليس لديك صلاحية تسجيل الطلبات" },
        { status: 403 }
      );
    }

    const formData = await request.formData();
    const payloadValue = formData.get("payload");

    if (typeof payloadValue !== "string") {
      return NextResponse.json(
        { error: "بيانات الطلب غير مكتملة" },
        { status: 400 }
      );
    }

    let payload: Payload;

    try {
      payload = JSON.parse(payloadValue) as Payload;
    } catch {
      return NextResponse.json(
        { error: "صيغة بيانات الطلب غير صحيحة" },
        { status: 400 }
      );
    }

    const {
      customerType,
      customerId,
      customerName,
      phone,
      email,
      address,
      taxNumber,
      customerNotes,
      proposedCompanyName,
      companyAlreadyEstablished,
      serviceId,
      serviceVariantId,
      description,
      priority,
      customerDocuments = [],
      questionAnswers = [],
      documentData = [],
    } = payload;

    if (!["NEW", "EXISTING"].includes(customerType)) {
      return NextResponse.json(
        { error: "نوع العميل غير صحيح" },
        { status: 400 }
      );
    }

    if (!clean(customerName)) {
      return NextResponse.json(
        { error: "اسم العميل أو الشركة مطلوب" },
        { status: 400 }
      );
    }

    if (!serviceId) {
      return NextResponse.json(
        { error: "يجب اختيار الخدمة" },
        { status: 400 }
      );
    }

    if (
      customerType === "NEW" &&
      !companyAlreadyEstablished &&
      !clean(proposedCompanyName)
    ) {
      return NextResponse.json(
        { error: "اسم الشركة المقترح مطلوب" },
        { status: 400 }
      );
    }

    if (customerType === "EXISTING" && !customerId) {
      return NextResponse.json(
        { error: "يجب البحث عن عميل حالي واختياره" },
        { status: 400 }
      );
    }

    if (
      payload.customerCategory &&
      ![
        "NEW_UNESTABLISHED",
        "NEW_ESTABLISHED",
        "EXISTING",
      ].includes(payload.customerCategory)
    ) {
      return NextResponse.json(
        { error: "فئة العميل غير صحيحة" },
        { status: 400 }
      );
    }

    if (
      payload.customerCategory === "EXISTING" &&
      customerType !== "EXISTING"
    ) {
      return NextResponse.json(
        { error: "بيانات فئة العميل غير متطابقة" },
        { status: 400 }
      );
    }

    const applicability = getApplicability(payload);

    // تحميل الخدمة ومتطلباتها وأسئلتها الفعالة من قاعدة البيانات.
    const service = await prisma.service.findFirst({

        id: serviceId,
        isActive: true,

          where: { isActive: true },
          orderBy: { sortOrder: "asc" },

          where: { isActive: true },

          where: {
            isActive: true,

            conditions: {
              include: {
                question: true,

    let selectedVariant: (typeof service.variants)[number] | null = null;

      if (!requirement || requirement.kind !== "DATA") {

    const missingDataRequirement = requirements.find(

        item.kind === "DATA" &&
        item.isRequired &&
        !submittedData.has(item.id)

    if (missingDataRequirement) {
      return NextResponse.json(
        { error: `يجب إدخال البيانات: ${missingDataRequirement.title}` },
        { status: 400 }
      );
    }

    // استقبال الملفات المرفوعة للمتطلبات الظاهرة من نوع DOCUMENT فقط.
    const submittedFiles: SubmittedFile[] = [];

    for (const [key, value] of formData.entries()) {
      if (!key.startsWith("document:")) continue;
      if (!(value instanceof File) || value.size === 0) continue;

      const requirementId = key.slice("document:".length);

      const requirement = requirements.find(
        (item) =>
          item.id === requirementId &&
          item.kind === "DOCUMENT"
      );

      if (!requirement) {
        return NextResponse.json(
          { error: "تم إرسال ملف لمستند غير مطلوب في هذه الحالة" },
          { status: 400 }
        );
      }

      if (value.size > MAX_FILE_SIZE) {
        return NextResponse.json(
          { error: `حجم الملف "${requirement.title}" أكبر من 10 ميجابايت` },
          { status: 400 }
        );
      }

      if (!ALLOWED_MIME_TYPES[value.type]) {
        return NextResponse.json(
          {
            error: `نوع الملف "${requirement.title}" غير مدعوم. ارفع PDF أو JPG أو PNG`,
          },
          { status: 400 }
        );
      }

      if (
        submittedFiles.some(
          (item) => item.requirementId === requirementId
        )
      ) {
        return NextResponse.json(
          { error: `تم إرسال أكثر من ملف للمتطلب "${requirement.title}"` },
          { status: 400 }
        );
      }

      submittedFiles.push({
        requirementId,
        file: value,
      });
    }

    // التحقق من بيانات المستندات الإضافية للعميل.
    for (const document of customerDocuments) {
      if (!clean(document.documentType)) {
        return NextResponse.json(
          { error: "يوجد مستند إضافي بدون نوع محدد" },
          { status: 400 }
        );
      }

      if (
        document.expiryDate &&
        Number.isNaN(Date.parse(document.expiryDate))
      ) {
        return NextResponse.json(
          { error: "تاريخ انتهاء أحد المستندات غير صحيح" },
          { status: 400 }
        );
      }
    }

    // التأكد من وجود العميل الحالي قبل بدء الحفظ.
    if (customerType === "EXISTING") {
      const existingCustomer = await prisma.customer.findUnique({
        where: { id: customerId },
      });

      if (!existingCustomer) {
        return NextResponse.json(
          { error: "العميل المختار غير موجود" },
          { status: 404 }
        );
      }
    }

    // منع إنشاء سجل عميل جديد مكرر.
    if (customerType === "NEW") {
      const duplicate = await prisma.customer.findFirst({
        where: clean(taxNumber)
          ? { taxNumber: clean(taxNumber) }
          : { name: clean(customerName) },
      });

      if (duplicate) {
        return NextResponse.json(
          {
            error:
              "يوجد عميل مسجل بهذه البيانات بالفعل. اختر «عميل حالي» وابحث عنه بدل إنشاء سجل مكرر.",
          },
          { status: 409 }
        );
      }
    }

    const result = await prisma.$transaction(async (tx) => {
      let customer;

      const customerData = {
        name: clean(customerName),
        phone: clean(phone) || null,
        email: clean(email) || null,
        address: clean(address) || null,
        taxNumber: clean(taxNumber) || null,
        notes: clean(customerNotes) || null,
      };

      if (customerType === "EXISTING") {
        customer = await tx.customer.update({
          where: { id: customerId },
          data: customerData,
        });
      } else {
        customer = await tx.customer.create({
          data: customerData,
        });

        createdCustomerId = customer.id;
      }

      if (customerDocuments.length > 0) {
        await tx.customerDocument.createMany({
          data: customerDocuments.map((document) => ({
            customerId: customer.id,
            documentType: clean(document.documentType),
            documentNumber: clean(document.documentNumber) || null,
            expiryDate: document.expiryDate
              ? new Date(`${document.expiryDate}T12:00:00`)
              : null,
            notes: clean(document.notes) || null,
          })),
        });
      }

      const requestNumber =
        `MAS-${new Date().toISOString().slice(0, 10).replace(/-/g, "")}-${randomUUID()
          .slice(0, 8)
          .toUpperCase()}`;

      const newRequest = await tx.request.create({
        data: {
          requestNumber,
          customerId: customer.id,
          createdById: user.id,
          currentDepartmentId: proceduresDepartment.id,
          serviceName: selectedVariant
            ? `${service.name} - ${selectedVariant.name}`
            : service.name,
          serviceId: service.id,
          serviceVariantId: selectedVariant?.id ?? null,
          customerType,
          proposedCompanyName: clean(proposedCompanyName) || null,
          companyAlreadyEstablished: Boolean(companyAlreadyEstablished),
          description: clean(description) || null,
          priority: ["NORMAL", "MEDIUM", "URGENT"].includes(priority || "")
            ? priority!
            : "NORMAL",
          status: "PENDING_REVIEW",
          submittedAt: new Date(),

          // حفظ نسخة من إجابات الأسئلة وقت تقديم الطلب.
          questionAnswers: {
            create: normalizedAnswers,
          },

          // حفظ نسخة من المتطلبات التي تنطبق على الطلب فقط.
          documentRequirements: {
            create: requirements.map((item) => {
              const dataValue =
                item.kind === "DATA"
                  ? submittedData.get(item.id) || null
                  : null;

              return {
                requirementId: item.id,
                requirementCode: item.code,
                title: item.title,
                details: item.details,
                condition: item.condition,
                isRequired: item.isRequired,
                applicability: item.applicability,
                kind: item.kind,
                value: dataValue,
                status: dataValue ? "RECEIVED" : "NOT_RECEIVED",
                receivedAt: dataValue ? new Date() : null,
                sortOrder: item.sortOrder,
              };
            }),
          },
        },
        include: {
          documentRequirements: true,
          questionAnswers: true,
          customer: true,
          service: true,
          serviceVariant: true,
        },
      });

      await tx.activityLog.create({
        data: {
          requestId: newRequest.id,
          userId: user.id,
          action: "REQUEST_SUBMITTED",
          details:
            "تم تسجيل الطلب وإرسال بياناته ومستنداته إلى مدير إدارة الإجراءات للمراجعة",
        },
      });

      return newRequest;
    });

    createdRequestId = result.id;

    // حفظ الملفات في مجلد خاص خارج public.
    const uploadDirectory = path.join(
      process.cwd(),
      "storage",
      "uploads",
      "requests",
      result.id
    );

    await mkdir(uploadDirectory, { recursive: true });

    for (const { requirementId, file } of submittedFiles) {
      const snapshot = result.documentRequirements.find(
        (item) => item.requirementId === requirementId
      );

      if (!snapshot) continue;

      const extension = ALLOWED_MIME_TYPES[file.type];
      const storedName = `${randomUUID()}${extension}`;

      const absolutePath = path.join(uploadDirectory, storedName);

      const relativePath = path.join(
        "storage",
        "uploads",
        "requests",
        result.id,
        storedName
      );

      const bytes = Buffer.from(await file.arrayBuffer());

      await writeFile(absolutePath, bytes);
      savedFiles.push(absolutePath);

      await prisma.requestDocumentRequirement.update({
        where: { id: snapshot.id },
        data: {
          filePath: relativePath,
          fileName: file.name,
          fileMimeType: file.type,
          fileSize: file.size,
          status: "RECEIVED",
          receivedAt: new Date(),
        },
      });
    }

    const updatedRequest = await prisma.request.findUnique({
      where: { id: result.id },
      include: {
        customer: true,
        documentRequirements: true,
        questionAnswers: true,
        service: true,
        serviceVariant: true,
      },
    });

    return NextResponse.json(
      {
        message: "تم تسجيل الطلب وإرساله للمراجعة بنجاح",
        request: updatedRequest,
      },
      { status: 201 }
    );
  } catch (error) {
    // حذف الطلب وملفاته في حالة فشل عملية الحفظ.
    if (createdRequestId) {
      await prisma.request
        .delete({ where: { id: createdRequestId } })
        .catch(() => undefined);

      await rm(
        path.join(
          process.cwd(),
          "storage",
          "uploads",
          "requests",
          createdRequestId
        ),
        { recursive: true, force: true }
      ).catch(() => undefined);
    } else {
      await Promise.all(
        savedFiles.map((file) =>
          rm(file, { force: true }).catch(() => undefined)
        )
      );
    }

    // لو تم إنشاء عميل جديد ثم فشل إنشاء الطلب، نحاول حذف سجل العميل الجديد.
   if (createdCustomerId) {
  await prisma.customer
    .delete({ where: { id: createdCustomerId } })
    .catch(() => undefined);
}
    console.error("Create request API error:", error);

    return NextResponse.json(
      { error: "حدث خطأ أثناء تسجيل الطلب أو حفظ المرفقات" },
      { status: 500 }
    );
  }
}

      requestId: id,
      assignedToId: user.id,

    select: {
      id: true,
    },

} else if (
  user.role === "SECRETARY" ||
  user.role === "CUSTOMER_SERVICE"
) {
  allowed = requestData.createdById === user.id;

export async function POST(request: Request) {
  try {
    const user = await getCurrentUser();

    if (!user || !user.isActive || !ALLOWED_ROLES.includes(user.role)) {
      return NextResponse.json(
        { error: "غير مصرح لك بتحويل الطلبات" },
        { status: 403 }
      );
    }

    const body = await request.json();

    const requestId = String(body.requestId || "").trim();
    const toDepartmentId = String(body.toDepartmentId || "").trim();
    const notes = String(body.notes || "").trim();

    if (!requestId || !toDepartmentId) {
      return NextResponse.json(
        { error: "اختر الطلب والقسم المستلم" },
        { status: 400 }
      );
    }

    const result = await prisma.$transaction(async (tx) => {
      const requestData = await tx.request.findUnique({
        where: { id: requestId },
        select: {
          id: true,
          requestNumber: true,
          currentDepartmentId: true,
          createdById: true,
        },
      });

      if (!requestData) {
        throw new Error("الطلب غير موجود");
      }

      if (
        user.role === "DEPARTMENT_MANAGER" &&
        requestData.currentDepartmentId !== user.departmentId
      ) {
        throw new Error("لا يمكنك تحويل طلب خارج قسمك");
      }

      if (
        (user.role === "SECRETARY" ||
          user.role === "CUSTOMER_SERVICE") &&
        requestData.createdById !== user.id
      ) {
        throw new Error("لا يمكنك تحويل طلب لم تقم بتسجيله");
      }

      if (requestData.currentDepartmentId === toDepartmentId) {
        throw new Error("الطلب موجود بالفعل في هذا القسم");
      }

      const destination = await tx.department.findUnique({
        where: { id: toDepartmentId },
        select: { id: true, name: true },
      });

      if (!destination) {
        throw new Error("القسم المستلم غير موجود");
      }

      const transfer = await tx.requestTransfer.create({
        data: {
          requestId,
          fromDepartmentId: requestData.currentDepartmentId,
          toDepartmentId,
          transferredById: user.id,
          notes: notes || null,
        },
      });

      await tx.request.update({
        where: { id: requestId },
        data: { currentDepartmentId: toDepartmentId },
      });

      await tx.activityLog.create({
        data: {
          requestId,
          userId: user.id,
          action: "REQUEST_TRANSFERRED",
          details: `تم تحويل الطلب ${requestData.requestNumber} إلى قسم ${destination.name}${notes ? `. ملاحظات: ${notes}` : ""}`,
        },
      });

      const destinationUsers = await tx.user.findMany({
        where: {
          departmentId: toDepartmentId,
          isActive: true,
          role: { in: ["DEPARTMENT_MANAGER", "EMPLOYEE"] },
        },
        select: { id: true },
      });

      if (destinationUsers.length > 0) {
        await tx.notification.createMany({
          data: destinationUsers.map((item) => ({
            userId: item.id,
            title: "طلب جديد وارد للقسم",
            message: `تم تحويل الطلب ${requestData.requestNumber} إلى قسم ${destination.name}`,
          })),
        });
      }

      return transfer;
    });

    return NextResponse.json(
      { message: "تم تحويل الطلب بنجاح", transfer: result },
      { status: 201 }
    );
  } catch (error) {
    console.error("POST transfers error:", error);

    const message =
      error instanceof Error ? error.message : "حدث خطأ أثناء تحويل الطلب";

    const status =
      [
        "الطلب غير موجود",
        "لا يمكنك تحويل طلب خارج قسمك",
        "لا يمكنك تحويل طلب لم تقم بتسجيله",
        "الطلب موجود بالفعل في هذا القسم",
        "القسم المستلم غير موجود",
      ].includes(message)
        ? 400
        : 500;

    return NextResponse.json({ error: message }, { status });
  }
}

    } else if (requestedDepartmentId) {

    const taskWhere: {
      departmentId?: string;
      assignedToId?: string;
    } = {};

    }

    if (user.role === "EMPLOYEE") {
      taskWhere.assignedToId = user.id;
    }

    const [departments, employees, taskTypes, tasks] =
      await Promise.all([
        user.role === "GENERAL_MANAGER"
          ? prisma.department.findMany({
              orderBy: { name: "asc" },
            })
          : prisma.department.findMany({
              where: { id: user.departmentId! },
            }),

        user.role === "EMPLOYEE"
          ? Promise.resolve([])
          : prisma.user.findMany({
              where: {
                isActive: true,
                role: "EMPLOYEE",
                ...(departmentId ? { departmentId } : {}),
              },
              select: {
                id: true,
                name: true,
                departmentId: true,
              },
              orderBy: { name: "asc" },
            }),

        prisma.taskType.findMany({
          where: {
            isActive: true,
            ...(departmentId ? { departmentId } : {}),
          },
          include: {
            department: {
              select: { id: true, name: true },
            },
          },
          orderBy: { name: "asc" },
        }),

        prisma.workflowTask.findMany({
          where: taskWhere,
          include: {
            request: {
              select: {
                id: true,
                requestNumber: true,
                serviceName: true,
                customer: {
                  select: { name: true },
                },
              },
            },
            department: {
              select: { id: true, name: true },
            },
            taskType: {
              select: { id: true, name: true },
            },
            assignedTo: {
              select: { id: true, name: true },
            },
            assignedBy: {
              select: { id: true, name: true },
            },
          },
          orderBy: { createdAt: "desc" },
          take: 100,
        }),
      ]);

    return NextResponse.json({
      departments,
      employees,
      taskTypes,
      suggestedTaskNames,
      tasks,
    });
  } catch (error) {
    console.error("Get tasks API error:", error);

    return NextResponse.json(
      { error: "حدث خطأ أثناء تحميل المهام" },
      { status: 500 }
    );
  }

      prisma.department.findUnique({
        where: { id: departmentId },
        select: { id: true, name: true },
      }),

      prisma.user.findUnique({
        where: { id: assignedToId },
        select: {
          id: true,
          name: true,
          role: true,
          departmentId: true,
          isActive: true,
        },
      }),
    ]);

    if (!requestData) {
      return NextResponse.json(
        { error: "الطلب غير موجود" },
        { status: 404 }
      );
    }

    if (!department) {
      return NextResponse.json(
        { error: "القسم غير موجود" },
        { status: 404 }
      );
    }

    // يجب تحويل الطلب للقسم قبل توزيع المهام عليه
    if (requestData.currentDepartmentId !== departmentId) {
      return NextResponse.json(
        { error: "يجب تحويل الطلب إلى هذا القسم أولاً" },
        { status: 400 }
      );
    }

    if (
      !employee ||
      !employee.isActive ||
      employee.role !== "EMPLOYEE" ||
      employee.departmentId !== departmentId
    ) {
      return NextResponse.json(
        { error: "الموظف غير نشط أو لا يتبع القسم المحدد" },
        { status: 400 }
      );
    }

    let expectedEndAt: Date | null = null;

    if (body.expectedEndAt) {
      expectedEndAt = new Date(body.expectedEndAt);

      if (Number.isNaN(expectedEndAt.getTime())) {
        return NextResponse.json(
          { error: "تاريخ الانتهاء المتوقع غير صحيح" },
          { status: 400 }
        );
      }
    }

    const task = await prisma.$transaction(async (tx) => {
      // إنشاء نوع المهمة تلقائيًا إذا لم يكن موجودًا بالقسم
      const taskType = await tx.taskType.upsert({
        where: {
          name_departmentId: {
            name: taskName,
            departmentId,
          },
        },
        update: {
          isActive: true,
        },
        create: {
          name: taskName,
          departmentId,
        },
      });

      const newTask = await tx.workflowTask.create({
        data: {
          requestId,
          departmentId,
          taskTypeId: taskType.id,
          assignedToId,
          assignedById: user.id,
          title: taskName,
          instructions: instructions || null,
          expectedEndAt,
          status: "PENDING",
        },
        include: {
          request: {
            select: {
              id: true,
              requestNumber: true,
            },
          },
          department: {
            select: {
              id: true,
              name: true,
            },
          },
          taskType: {
            select: {
              id: true,
              name: true,
            },
          },
          assignedTo: {
            select: {
              id: true,
              name: true,
            },
          },
        },
      });

      await tx.activityLog.create({
        data: {
          requestId,
          taskId: newTask.id,
          userId: user.id,
          action: "TASK_ASSIGNED",
          details: `تم توزيع مهمة "${taskName}" على ${employee.name} بقسم ${department.name}`,
        },
      });

      await tx.notification.create({
        data: {
          userId: assignedToId,
          title: "مهمة جديدة",
          message: `تم إسناد مهمة "${taskName}" إليك للطلب رقم ${requestData.requestNumber}`,
        },
      });

      return newTask;
    });

    return NextResponse.json(
      {
        message: "تم إنشاء المهمة وتوزيعها بنجاح",
        task,
      },
      { status: 201 }
    );
  } catch (error) {
    console.error("Create task API error:", error);

    return NextResponse.json(
      { error: "حدث خطأ أثناء إنشاء المهمة" },
      { status: 500 }
    );
  }

import { NextRequest, NextResponse } from "next/server";

async function canAccessTask(
  taskId: string,
  user: {
    id: string;
    role: string;
    departmentId: string | null;
  }
) {
  const task = await prisma.workflowTask.findUnique({
    where: { id: taskId },
    select: {
      id: true,
      requestId: true,
      departmentId: true,
      assignedToId: true,
      status: true,
      procedureTemplateId: true,
      startedAt: true,
      steps: {
        select: {
          id: true,
        },
      },
    },
  });

  if (!task) return null;

  const allowed =
    user.role === "GENERAL_MANAGER" ||
    (user.role === "DEPARTMENT_MANAGER" &&
      user.departmentId === task.departmentId) ||
    (user.role === "EMPLOYEE" &&
      user.id === task.assignedToId);

  return allowed ? task : null;
}

export async function POST(request: NextRequest) {
  try {
    const user = await getCurrentUser();

    if (!user) {
      return NextResponse.json(
        { error: "يجب تسجيل الدخول أولاً" },
        { status: 401 }
      );
    }

    if (
      !["GENERAL_MANAGER", "DEPARTMENT_MANAGER", "EMPLOYEE"].includes(
        user.role
      )
    ) {
      return NextResponse.json(
        { error: "ليس لديك صلاحية تنفيذ المهام" },
        { status: 403 }
      );
    }

    const body = await request.json();

    const {
      action,
      taskId,
      stepId,
      stepStatus,
      stepNotes,
      amount,
      expenseDescription,
    } = body;

    if (!taskId || !action) {
      return NextResponse.json(
        { error: "بيانات العملية غير مكتملة" },
        { status: 400 }
      );
    }

    const task = await canAccessTask(taskId, user);

    if (!task) {
      return NextResponse.json(
        {
          error: "المهمة غير موجودة أو ليس لديك صلاحية الوصول إليها",
        },
        { status: 404 }
      );
    }

    if (action === "START") {
      if (task.status === "COMPLETED") {
        return NextResponse.json(
          { error: "لا يمكن بدء مهمة مكتملة" },
          { status: 400 }
        );
      }

      if (task.status !== "IN_PROGRESS") {
        if (task.steps.length === 0 && task.procedureTemplateId) {
          const templateSteps =
            await prisma.procedureStepTemplate.findMany({
              where: {
                procedureTemplateId: task.procedureTemplateId,
              },
              orderBy: {
                order: "asc",
              },
            });

          if (templateSteps.length > 0) {
            await prisma.taskStep.createMany({
              data: templateSteps.map((step) => ({
                taskId,
                templateStepId: step.id,
                title: step.title,
                notes: step.description,
                order: step.order,
              })),
            });
          }
        }

        await prisma.workflowTask.update({
          where: {
            id: taskId,
          },
          data: {
            status: "IN_PROGRESS",
            startedAt: task.startedAt || new Date(),
          },
        });

        await prisma.activityLog.create({
          data: {
            requestId: task.requestId,
            taskId,
            userId: user.id,
            action: "TASK_STARTED",
            details: "تم بدء تنفيذ المهمة",
          },
        });
      }

      return NextResponse.json({
        success: true,
        message: "تم بدء تنفيذ المهمة",
      });
    }

    if (action === "UPDATE_STEP") {
      if (!stepId) {
        return NextResponse.json(
          { error: "يجب تحديد خطوة التنفيذ" },
          { status: 400 }
        );
      }

      if (
        !["PENDING", "IN_PROGRESS", "COMPLETED"].includes(stepStatus)
      ) {
        return NextResponse.json(
          { error: "حالة الخطوة غير صحيحة" },
          { status: 400 }
        );
      }

      if (task.status === "COMPLETED") {
        return NextResponse.json(
          { error: "المهمة مكتملة بالفعل" },
          { status: 400 }
        );
      }

      const step = await prisma.taskStep.findFirst({
        where: {
          id: stepId,
          taskId,
        },
      });

      if (!step) {
        return NextResponse.json(
          { error: "خطوة التنفيذ غير موجودة" },
          { status: 404 }
        );
      }

      await prisma.taskStep.update({
        where: {
          id: stepId,
        },
        data: {
          status: stepStatus,
          notes:
            typeof stepNotes === "string"
              ? stepNotes
              : step.notes,
          completedAt:
            stepStatus === "COMPLETED"
              ? new Date()
              : null,
        },
      });

      await prisma.activityLog.create({
        data: {
          requestId: task.requestId,
          taskId,
          userId: user.id,
          action: "TASK_STEP_UPDATED",
          details: `تم تحديث خطوة التنفيذ: ${step.title} - ${stepStatus}`,
        },
      });

      return NextResponse.json({
        success: true,
        message: "تم تحديث خطوة التنفيذ",
      });
    }

    if (action === "ADD_EXPENSE") {
      if (task.status === "COMPLETED") {
        return NextResponse.json(
          { error: "لا يمكن إضافة مصروف لمهمة مكتملة" },
          { status: 400 }
        );
      }

      const parsedAmount = Number(amount);

      if (
        !Number.isFinite(parsedAmount) ||
        parsedAmount <= 0 ||
        typeof expenseDescription !== "string" ||
        !expenseDescription.trim()
      ) {
        return NextResponse.json(
          { error: "أدخل مبلغًا صحيحًا ووصفًا للمصروف" },
          { status: 400 }
        );
      }

      await prisma.expense.create({
        data: {
          requestId: task.requestId,
          taskId,
          employeeId: user.id,
          amount: parsedAmount,
          description: expenseDescription.trim(),
        },
      });

      await prisma.activityLog.create({
        data: {
          requestId: task.requestId,
          taskId,
          userId: user.id,
          action: "TASK_EXPENSE_ADDED",
          details: `تم تسجيل مصروف بقيمة ${parsedAmount}: ${expenseDescription.trim()}`,
        },
      });

      return NextResponse.json({
        success: true,
        message: "تم تسجيل المصروف",
      });
    }

    if (action === "COMPLETE") {
      if (task.status === "COMPLETED") {
        return NextResponse.json({
          success: true,
          message: "المهمة مكتملة بالفعل",
        });
      }

      const pendingSteps = await prisma.taskStep.count({
        where: {
          taskId,
          status: {
            not: "COMPLETED",
          },
        },
      });

      if (pendingSteps > 0) {
        return NextResponse.json(
          { error: "يجب إكمال جميع خطوات المهمة أولاً" },
          { status: 400 }
        );
      }

      await prisma.workflowTask.update({
        where: {
          id: taskId,
        },
        data: {
          status: "COMPLETED",
          completedAt: new Date(),
          startedAt: task.startedAt || new Date(),
        },
      });

      await prisma.activityLog.create({
        data: {
          requestId: task.requestId,
          taskId,
          userId: user.id,
          action: "TASK_COMPLETED",
          details: "تم الانتهاء من تنفيذ المهمة",
        },
      });

      return NextResponse.json({
        success: true,
        message: "تم إكمال المهمة بنجاح",
      });
    }

    return NextResponse.json(
      { error: "العملية المطلوبة غير معروفة" },
      { status: 400 }
    );
  } catch (error) {
    console.error("Execution POST error:", error);

    return NextResponse.json(
      { error: "حدث خطأ أثناء تنفيذ العملية" },
      { status: 500 }
    );
  }
}

    if (!task) {
      return NextResponse.json(
        { error: "المهمة غير موجودة" },
        { status: 404 }
      );
    }

    if (
      user.role === "DEPARTMENT_MANAGER" &&
      user.departmentId !== task.departmentId
    ) {
      return NextResponse.json(
        { error: "لا يمكنك تسليم مهمة تابعة لقسم آخر" },
        { status: 403 }
      );
    }

    if (task.status !== "APPROVED") {
      return NextResponse.json(
        { error: "لا يمكن تسليم المهمة إلا بعد اعتمادها من المراجعة" },
        { status: 400 }
      );
    }

    const details = JSON.stringify({
      deliveryMethod: deliveryMethod.trim(),
      deliveryOutputs: deliveryOutputs.trim(),
      deliveryDate: parsedDate.toISOString(),
      deliveryNotes:
        typeof deliveryNotes === "string"
          ? deliveryNotes.trim()
          : "",
      deliveredBy: user.name,
    });

    await prisma.$transaction([
      prisma.workflowTask.update({
        where: {
          id: taskId,
        },
        data: {
          status: "DELIVERED",
        },
      }),
      prisma.activityLog.create({
        data: {
          requestId: task.requestId,
          taskId,
          userId: user.id,
          action: "TASK_DELIVERED",
          details,
        },
      }),
    ]);

    const remainingTasks = await prisma.workflowTask.count({
      where: {
        requestId: task.requestId,
        status: {
          not: "DELIVERED",
        },
      },
    });

    if (remainingTasks === 0) {
      await prisma.request.update({
        where: {
          id: task.requestId,
        },
        data: {
          status: "DELIVERED",
        },
      });
    }

    return NextResponse.json({
      success: true,
      message:
        remainingTasks === 0
          ? "تم تسليم جميع مهام الطلب وتحديث حالة الطلب"
          : "تم تسجيل تسليم المهمة بنجاح",
    });
  } catch (error) {
    console.error("Delivery POST error:", error);

    return NextResponse.json(
      { error: "حدث خطأ أثناء تسجيل التسليم" },
      { status: 500 }
    );
  }

export async function POST(request: NextRequest) {
  try {
    const user = await getCurrentUser();

    if (!user) {
      return NextResponse.json(
        { error: "يجب تسجيل الدخول أولاً" },
        { status: 401 }
      );
    }

    if (!allowedRoles.includes(user.role)) {
      return NextResponse.json(
        { error: "ليس لديك صلاحية إغلاق الطلبات" },
        { status: 403 }
      );
    }

    const body = await request.json();

    const {
      requestId,
      clientReceived,
      closureDate,
      clientRating,
      closureNotes,
    } = body;

    if (
      typeof requestId !== "string" ||
      !requestId ||
      clientReceived !== "تم الاستلام" ||
      typeof closureDate !== "string" ||
      !closureDate
    ) {
      return NextResponse.json(
        { error: "يجب تأكيد استلام العميل وإدخال تاريخ الإغلاق" },
        { status: 400 }
      );
    }

    const parsedDate = new Date(closureDate);

    if (Number.isNaN(parsedDate.getTime())) {
      return NextResponse.json(
        { error: "تاريخ الإغلاق غير صحيح" },
        { status: 400 }
      );
    }

    const existingRequest = await prisma.request.findUnique({
      where: {
        id: requestId,
      },
      select: {
        id: true,
        requestNumber: true,
        status: true,
        currentDepartmentId: true,
      },
    });

    if (!existingRequest) {
      return NextResponse.json(
        { error: "الطلب غير موجود" },
        { status: 404 }
      );
    }

    if (
      user.role === "DEPARTMENT_MANAGER" &&
      user.departmentId !== existingRequest.currentDepartmentId
    ) {
      return NextResponse.json(
        { error: "لا يمكنك إغلاق طلب تابع لقسم آخر" },
        { status: 403 }
      );
    }

    if (existingRequest.status === "CLOSED") {
      return NextResponse.json(
        { error: "تم إغلاق هذا الطلب بالفعل" },
        { status: 400 }
      );
    }

    if (existingRequest.status !== "DELIVERED") {
      return NextResponse.json(
        { error: "يجب تسليم الطلب أولاً قبل إغلاقه" },
        { status: 400 }
      );
    }

    const [totalTasks, unfinishedTasks] = await Promise.all([
      prisma.workflowTask.count({
        where: {
          requestId,
        },
      }),
      prisma.workflowTask.count({
        where: {
          requestId,
          status: {
            not: "DELIVERED",
          },
        },
      }),
    ]);

    if (totalTasks === 0 || unfinishedTasks > 0) {
      return NextResponse.json(
        { error: "لا يمكن إغلاق الطلب قبل تسليم جميع مهامه" },
        { status: 400 }
      );
    }

    const details = JSON.stringify({
      clientReceived,
      closureDate: parsedDate.toISOString(),
      clientRating:
        typeof clientRating === "string" ? clientRating : "",
      closureNotes:
        typeof closureNotes === "string" ? closureNotes.trim() : "",
      closedBy: user.name,
    });

    await prisma.$transaction([
      prisma.request.update({
        where: {
          id: requestId,
        },
        data: {
          status: "CLOSED",
          closedAt: parsedDate,
        },
      }),
      prisma.activityLog.create({
        data: {
          requestId,
          userId: user.id,
          action: "REQUEST_CLOSED",
          details,
        },
      }),
    ]);

    return NextResponse.json({
      success: true,
      message: "تم إغلاق الطلب بنجاح",
    });
  } catch (error) {
    console.error("Closure POST error:", error);

    return NextResponse.json(
      { error: "حدث خطأ أثناء إغلاق الطلب" },
      { status: 500 }
    );
  }
}

function formatFileSize(size: number) {
  if (size < 1024 * 1024) {
    return `${(size / 1024).toFixed(0)} كيلوبايت`;

function parseQuestionOptions(optionsJson: string | null): string[] {
  if (!optionsJson) return [];

  try {
    const parsed = JSON.parse(optionsJson);

    if (!Array.isArray(parsed)) return [];

    return parsed
      .map((item) => {
        if (typeof item === "string") return item;
        if (item && typeof item.label === "string") return item.label;
        if (item && typeof item.value === "string") return item.value;
        return "";
      })
      .filter(Boolean);
  } catch {
    return [];
  }
}

function requirementMatchesAnswers(
  requirement: DocumentRequirement,
  answers: Record<string, string>
) {
  const conditions = requirement.conditions || [];

  // المتطلب الذي ليس له شروط يظهر دائمًا.
  if (conditions.length === 0) return true;

  // الشروط داخل المجموعة الواحدة تعمل بمنطق AND،
  // والمجموعات المختلفة تعمل بمنطق OR.
  const groups = new Map<string, RequirementCondition[]>();

  for (const condition of conditions) {
    const groupKey = condition.groupKey || "DEFAULT";
    const current = groups.get(groupKey) || [];
    current.push(condition);
    groups.set(groupKey, current);
  }

  return Array.from(groups.values()).some((group) =>
    group.every(
      (condition) =>
        answers[condition.question.code] === condition.expectedValue
    )
  );
}

  closureDate: string;

  const [selectedId, setSelectedId] = useState("");
  const [form, setForm] = useState<ClosureForm>(emptyForm);
  const [filter, setFilter] = useState("ALL");
  const [message, setMessage] = useState("");

        cache: "no-store",
