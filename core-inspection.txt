import "server-only";
import { SignJWT, jwtVerify } from "jose";
import { cookies } from "next/headers";
import { redirect } from "next/navigation";
import { prisma } from "./prisma";

const COOKIE_NAME = "masar_session";
const secret = new TextEncoder().encode(process.env.AUTH_SECRET);

if (!process.env.AUTH_SECRET || process.env.AUTH_SECRET.length < 32) {
  throw new Error("AUTH_SECRET must be at least 32 characters");
}

type SessionUser = {
  id: string;
  role: string;
  departmentId: string | null;
};

export async function createSession(user: SessionUser) {
  const token = await new SignJWT({
    role: user.role,
    departmentId: user.departmentId,
  })
    .setProtectedHeader({ alg: "HS256" })
    .setSubject(user.id)
    .setIssuedAt()
    .setExpirationTime("8h")
    .sign(secret);

  const cookieStore = await cookies();

  cookieStore.set(COOKIE_NAME, token, {
    httpOnly: true,
    secure: process.env.NODE_ENV === "production",
    sameSite: "lax",
    path: "/",
    maxAge: 60 * 60 * 8,
  });
}

export async function deleteSession() {
  const cookieStore = await cookies();
  cookieStore.delete(COOKIE_NAME);
}

export async function getCurrentUser() {
  const cookieStore = await cookies();
  const token = cookieStore.get(COOKIE_NAME)?.value;

  if (!token) return null;

  try {
    const { payload } = await jwtVerify(token, secret);

    if (!payload.sub) return null;

    const user = await prisma.user.findUnique({
      where: { id: payload.sub },
      select: {
        id: true,
        name: true,
        email: true,
        role: true,
        departmentId: true,
        isActive: true,
        department: {
          select: { name: true },
        },
      },
    });

    if (!user || !user.isActive) return null;

    return user;
  } catch {
    return null;
  }
}

export async function requireUser() {
  const user = await getCurrentUser();

  if (!user) {
    redirect("/login");
  }

  return user;
}

import "server-only";
import { redirect } from "next/navigation";
import { getCurrentUser } from "./auth";

export const ROLES = {
  GENERAL_MANAGER: "GENERAL_MANAGER",
  DEPARTMENT_MANAGER: "DEPARTMENT_MANAGER",
  EMPLOYEE: "EMPLOYEE",
} as const;

export async function requireRole(allowedRoles: string[]) {
  const user = await getCurrentUser();

  if (!user) {
    redirect("/login");
  }

  if (!allowedRoles.includes(user.role)) {
    redirect("/unauthorized");
  }

  return user;
}

export async function requireGeneralManager() {
  return requireRole([ROLES.GENERAL_MANAGER]);
}

export async function requireDepartmentManager() {
  return requireRole([
    ROLES.GENERAL_MANAGER,
    ROLES.DEPARTMENT_MANAGER,
  ]);
}

export async function requireEmployee() {
  return requireRole([
    ROLES.GENERAL_MANAGER,
    ROLES.DEPARTMENT_MANAGER,
    ROLES.EMPLOYEE,
  ]);
}

export async function canAccessDepartment(departmentId: string) {
  const user = await getCurrentUser();

  if (!user) return false;

  if (user.role === ROLES.GENERAL_MANAGER) return true;

  return user.departmentId === departmentId;
}

import { NextResponse } from "next/server";
import { compare } from "bcryptjs";
import { prisma } from "@/lib/prisma";
import { createSession } from "@/lib/auth";

export async function POST(request: Request) {
  try {
    const body = await request.json();

    const email =
      typeof body.email === "string" ? body.email.trim().toLowerCase() : "";

    const password =
      typeof body.password === "string" ? body.password : "";

    if (!email || !password) {
      return NextResponse.json(
        { message: "أدخل البريد الإلكتروني وكلمة المرور" },
        { status: 400 }
      );
    }

    const user = await prisma.user.findUnique({
      where: { email },
    });

    if (!user || !user.isActive) {
      return NextResponse.json(
        { message: "بيانات الدخول غير صحيحة" },
        { status: 401 }
      );
    }

    const passwordIsValid = await compare(
      password,
      user.passwordHash
    );

    if (!passwordIsValid) {
      return NextResponse.json(
        { message: "بيانات الدخول غير صحيحة" },
        { status: 401 }
      );
    }

    await createSession({
      id: user.id,
      role: user.role,
      departmentId: user.departmentId,
    });

    return NextResponse.json({
      success: true,
      message: "تم تسجيل الدخول بنجاح",
      user: {
        id: user.id,
        name: user.name,
        email: user.email,
        role: user.role,
        departmentId: user.departmentId,
      },
    });
  } catch (error) {
    console.error("Login error:", error);

    return NextResponse.json(
      { message: "حدث خطأ أثناء تسجيل الدخول" },
      { status: 500 }
    );
  }
}

  }

    setRequest((current) => ({ ...current, [field]: value }));
  }

  function addDocument() {
    setDocuments((current) => [
      ...current,
      { documentType: "", documentNumber: "", expiryDate: "" },
    ]);
  }

  function updateDocument(
    index: number,
    field: string,
    value: string
  ) {
    setDocuments((current) =>
      current.map((document, i) =>
        i === index ? { ...document, [field]: value } : document
      )
    );
  }

  return (
    <div className="form-card">
      <h2>بيانات العميل</h2>
      <p>حدد إذا كانت الشركة تتعامل معنا لأول مرة أم لديها ملف سابق.</p>

      <div className="form-grid">
        <div className="form-field">
          <label>نوع العميل</label>
          <select
            value={customerType}
            onChange={(e) => {
              const value = e.target.value as "new" | "existing";
              setCustomerType(value);
              setSelectedCustomer(null);
              setSearch("");
              setResults([]);
              setError("");
              setCustomer({
                name: "",
                phone: "",
                email: "",
                address: "",
                taxNumber: "",
                notes: "",
              });
            }}
          >
            <option value="new">عميل جديد</option>
            <option value="existing">عميل حالي</option>
          </select>
        </div>

        {customerType === "existing" && (
          <div className="form-field">
            <label htmlFor="customerSearch">
              ابحث باسم الشركة أو رقم الهاتف أو الرقم الضريبي
            </label>
            <input
              id="customerSearch"
              value={search}
              onChange={(e) => searchCustomers(e.target.value)}
              placeholder="اكتب اسم الشركة أو رقم الهاتف..."
            />

            {searching && <p>جاري البحث...</p>}

            {results.length > 0 && (
              <div className="customer-search-results">
                {results.map((item) => (
                  <button
                    type="button"
                    key={item.id}
                    onClick={() => chooseCustomer(item)}
                    className="customer-result"
                  >
                    <strong>{item.name}</strong>
                    <span>
                      {item.taxNumber || "لا يوجد رقم ضريبي"} —{" "}
                      {item.phone || "لا يوجد هاتف"}
                    </span>
                  </button>
                ))}
              </div>
            )}

            {search.trim().length >= 2 &&
              !searching &&
              results.length === 0 &&
              !selectedCustomer && (
                <p>لم يتم العثور على عميل مطابق.</p>
              )}
          </div>
        )}
      </div>

      {selectedCustomer && (
        <div className="form-card" style={{ marginTop: 20 }}>
          <h3>ملف العميل المحفوظ</h3>
          <p>
            تم تحميل بيانات الشركة من قاعدة البيانات. يمكنك تعديل البيانات
            قبل حفظ الطلب.
          </p>

          <div className="form-grid">
            {[
              ["name", "اسم الشركة"],
              ["phone", "رقم الهاتف"],
              ["email", "البريد الإلكتروني"],
              ["address", "العنوان"],
              ["taxNumber", "الرقم الضريبي"],
            ].map(([field, label]) => (
              <div className="form-field" key={field}>
                <label>{label}</label>
                <input
                  value={customer[field as keyof typeof customer]}
                  onChange={(e) => updateCustomer(field, e.target.value)}
                />
              </div>
            ))}
          </div>

          <h3 style={{ marginTop: 24 }}>المستندات المسجلة للشركة</h3>

          {selectedCustomer.documents.length === 0 ? (
            <p>لا توجد مستندات محفوظة لهذا العميل حتى الآن.</p>
          ) : (
            <div className="form-grid">
              {selectedCustomer.documents.map((document) => (
                <div className="form-field" key={document.id}>
                  <strong>{document.documentType}</strong>
                  <span>
                    رقم المستند: {document.documentNumber || "غير مسجل"}
                  </span>
                  <span>
                    تاريخ الانتهاء:{" "}
                    {document.expiryDate
                      ? new Date(document.expiryDate).toLocaleDateString("ar-EG")
                      : "غير محدد"}
                  </span>
                  {document.filePath && (
                    <a href={document.filePath} target="_blank" rel="noreferrer">
                      فتح الملف المحفوظ
                    </a>
                  )}
                </div>
              ))}
            </div>
          )}
        </div>
      )}

      {(customerType === "new" || selectedCustomer) && (
        <>
          <h3 style={{ marginTop: 24 }}>
            {customerType === "new" ? "بيانات الشركة الجديدة" : "تعديل بيانات الشركة"}
          </h3>

          <div className="form-grid">
            {[
              ["name", "اسم الشركة"],
              ["phone", "رقم الهاتف"],
              ["email", "البريد الإلكتروني"],
              ["address", "العنوان"],
              ["taxNumber", "الرقم الضريبي"],
            ].map(([field, label]) => (
              <div className="form-field" key={field}>
                <label>{label}</label>
                <input
                  required={field === "name"}
                  value={customer[field as keyof typeof customer]}
                  onChange={(e) => updateCustomer(field, e.target.value)}
                  placeholder={`أدخل ${label}`}
                />
              </div>
            ))}

            <div className="form-field full-width">
              <label>ملاحظات عن العميل</label>
              <textarea
                rows={3}
                value={customer.notes}
                onChange={(e) => updateCustomer("notes", e.target.value)}
              />
            </div>
          </div>

          <h3 style={{ marginTop: 24 }}>إضافة مستندات للعميل</h3>

          {documents.map((document, index) => (
            <div className="form-grid" key={index}>
              <div className="form-field">
                <label>نوع المستند</label>
                <select
                  value={document.documentType}
                  onChange={(e) =>
                    updateDocument(index, "documentType", e.target.value)
                  }
                >
                  <option value="">اختر نوع المستند</option>
                  {documentTypes.map((type) => (
                    <option key={type} value={type}>{type}</option>
                  ))}
                </select>
              </div>

              <div className="form-field">
                <label>رقم المستند</label>
                <input
                  value={document.documentNumber}
                  onChange={(e) =>
                    updateDocument(index, "documentNumber", e.target.value)
                  }
                />
              </div>

              <div className="form-field">
                <label>تاريخ الانتهاء</label>
                <input
                  type="date"
                  value={document.expiryDate}
                  onChange={(e) =>
                    updateDocument(index, "expiryDate", e.target.value)
                  }
                />
              </div>
            </div>
          ))}

          <button
            type="button"
            className="secondary-button"
            onClick={addDocument}
          >
            + إضافة مستند
          </button>
        </>
      )}

      <hr style={{ margin: "28px 0" }} />

      <h2>بيانات الطلب الجديد</h2>

      <div className="form-grid">
        <div className="form-field">
          <label>الخدمة المطلوبة</label>
          <input
            value={request.serviceName}
            onChange={(e) => updateRequest("serviceName", e.target.value)}
            placeholder="اكتب الخدمة المطلوبة"
          />
        </div>

        <div className="form-field">
          <label>تاريخ استلام الطلب</label>
          <input
            type="date"
            value={request.receivedAt}
            onChange={(e) => updateRequest("receivedAt", e.target.value)}
          />
        </div>

        <div className="form-field">
          <label>الأولوية</label>
          <select
            value={request.priority}
            onChange={(e) => updateRequest("priority", e.target.value)}
          >
            <option value="NORMAL">عادي</option>
            <option value="MEDIUM">متوسط</option>
            <option value="URGENT">عاجل</option>
          </select>
        </div>

        <div className="form-field full-width">
          <label>وصف الطلب</label>
          <textarea
            rows={4}
            value={request.description}
            onChange={(e) => updateRequest("description", e.target.value)}
          />
        </div>
      </div>

      {error && <p style={{ color: "red" }}>{error}</p>}

      <p style={{ marginTop: 20, color: "#64748b" }}>
        تم تجهيز البحث وعرض ملف العميل. الخطوة التالية هي ربط زر الحفظ
        بقاعدة البيانات ورفع ملفات المستندات فعليًا.
      </p>
    </div>
  );
}
